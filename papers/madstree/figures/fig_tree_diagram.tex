% Standalone source of the massive three-vertex chain of the first application
% example (Sec. 5), redrawn element by element after the original
% three_vertex_tree figure: hatched-circle time vertices on a horizontal axis,
% two schematic external legs per vertex ending in open squares with the summed
% energy E_v beside the fan, teal internal propagators carrying the
% parenthesised (momentum, mass) labels above the line, and the vertex labels
% V_v(\tau_v,p_v) below the vertices.
% Compile: pdflatex fig_tree_diagram.tex   (included by paper.tex)
\documentclass[tikz,border=6pt]{standalone}
\usetikzlibrary{patterns.meta}
\definecolor{propteal}{RGB}{0,166,172}
\tikzset{
  vtx/.style={circle, draw=black, line width=1.0pt, fill=white,
    postaction={pattern={Lines[angle=45, distance=1.3pt, line width=0.45pt]},
      pattern color=black},
    minimum size=#1, inner sep=0pt, outer sep=0pt},
  vtx/.default=4.2pt,
  leg/.style={draw=black, line width=0.9pt},
  sq/.style={draw=black, line width=0.9pt, fill=white, minimum size=3.0pt,
    inner sep=0pt, outer sep=0pt},
  prop/.style={line width=1.2pt},
}
\begin{document}
\begin{tikzpicture}[every node/.style={inner sep=1.5pt}]
  \coordinate (vL) at (0,0);
  \coordinate (vM) at (2.22,0);
  \coordinate (vR) at (4.54,0);
  % External legs: two per vertex, open squares at the tips.
  \draw[leg] (vL) -- ++(149:1.52) node[sq] {};
  \draw[leg] (vL) -- ++(200:1.45) node[sq] {};
  \draw[leg] (vM) -- ++(120:1.85) node[sq] {};
  \draw[leg] (vM) -- ++(60:1.85)  node[sq] {};
  \draw[leg] (vR) -- ++(31:1.52)  node[sq] {};
  \draw[leg] (vR) -- ++(-20:1.45) node[sq] {};
  % Summed external energies beside the fans.
  \node at (-1.02,1.06) {\small $E_1$};
  \node at (2.22,2.10)  {\small $E_2$};
  \node at (5.56,1.06)  {\small $E_3$};
  % Internal massive propagators: teal lines with parenthesised labels above.
  \draw[prop,propteal] (vL) -- (vM);
  \draw[prop,propteal] (vM) -- (vR);
  \node[propteal] at (1.11,0.28) {\small $(s_1,\widetilde\nu_1)$};
  \node[propteal] at (3.38,0.28) {\small $(s_2,\widetilde\nu_2)$};
  % Vertex labels below the vertices.
  \node[anchor=north] at (0,-0.20)    {\small $V_L(\tau_L,p_1)$};
  \node[anchor=north] at (2.22,-0.20) {\small $V_M(\tau_M,p_2)$};
  \node[anchor=north] at (4.54,-0.20) {\small $V_R(\tau_R,p_3)$};
  % Vertices last, on top of the line ends.
  \node[vtx] at (vL) {};
  \node[vtx] at (vM) {};
  \node[vtx] at (vR) {};
\end{tikzpicture}
\end{document}
