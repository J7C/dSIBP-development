% Standalone source of the four contracted-sector diagrams of the three-vertex
% chain examples (Sec. 5). Merged components are drawn as single dots whose size
% grows with the number of merged vertices; the contracted propagator is absent.
% Panel labels use the sector names of the examples: Top, LeftPinch, RightPinch,
% DoublePinch.
% Compile: pdflatex fig_tree_sectors.tex   (included by paper.tex)
\documentclass[tikz,border=6pt]{standalone}
\usetikzlibrary{calc,positioning}
\definecolor{propblue}{RGB}{0,161,168}
\begin{document}
\begin{tikzpicture}[
  every node/.style={inner sep=1.5pt},
  one/.style={circle,fill=black,draw=black,minimum size=6pt,inner sep=0pt},
  two/.style={circle,fill=black,draw=black,minimum size=9pt,inner sep=0pt},
  three/.style={circle,fill=black,draw=black,minimum size=12pt,inner sep=0pt},
  lab/.style={font=\small,anchor=north},
]
  % ============ Top sector ============
  \begin{scope}[shift={(0,0)}]
    \node[one] (TL) at (0,0) {};
    \node[one] (TM) at (2.0,0) {};
    \node[one] (TR) at (4.0,0) {};
    \draw[->,thick,propblue] (TL) -- (TM);
    \draw[->,thick,propblue] (TM) -- (TR);
    \node[lab] at (2.0,-1.15) {Top sector};
  \end{scope}
  % ============ LeftPinch sector ============
  \begin{scope}[shift={(6.2,0)}]
    \node[two] (PL) at (0,0) {};
    \node[one] (PR) at (2.6,0) {};
    \draw[->,thick,propblue] (PL) -- (PR);
    \node[lab] at (1.3,-1.15) {LeftPinch sector};
  \end{scope}
  % ============ RightPinch sector ============
  \begin{scope}[shift={(0,-2.5)}]
    \node[one] (QL) at (0,0) {};
    \node[two] (QR) at (2.6,0) {};
    \draw[->,thick,propblue] (QL) -- (QR);
    \node[lab] at (1.3,-1.15) {RightPinch sector};
  \end{scope}
  % ============ DoublePinch sector ============
  \begin{scope}[shift={(6.2,-2.5)}]
    \node[three] (DD) at (0,0) {};
    \node[lab] at (0,-1.15) {DoublePinch sector};
  \end{scope}
\end{tikzpicture}
\end{document}
