% Standalone source of the contracted-sector diagrams of the bubble time kernel
% (Sec. 5): the Top sector keeps the circular loop; pinching one propagator
% merges the two vertices --- the larger dot --- and leaves the other propagator
% as a circular tadpole on the single merged component. Colors follow the main
% figure: propagator 1 teal-green (upper half), propagator 2 purple (lower
% half); Pinch1 contracts propagator 1 and keeps propagator 2, Pinch2 the
% mirror image.
% Compile: pdflatex fig_bubble_sectors.tex   (included by paper.tex)
\documentclass[tikz,border=6pt]{standalone}
\usetikzlibrary{calc,positioning}
\definecolor{propgreen}{RGB}{0,161,150}
\definecolor{proppurple}{RGB}{121,90,188}
\begin{document}
\begin{tikzpicture}[
  every node/.style={inner sep=1.5pt},
  dot/.style={circle,fill=black,draw=black,minimum size=6pt,inner sep=0pt},
  bigdot/.style={circle,fill=black,draw=black,minimum size=9pt,inner sep=0pt},
  lab/.style={font=\small,anchor=north},
]
  % ============ Top sector ============
  \begin{scope}[shift={(0,0)}]
    \node[dot] (v1) at (0,0) {};
    \node[dot] (v2) at (2.6,0) {};
    \draw[->,thick,propgreen]  (v1) arc[start angle=180,end angle=0,radius=1.3];
    \draw[->,thick,proppurple] (v1) arc[start angle=180,end angle=360,radius=1.3];
    \node[lab] at (1.3,-1.65) {Top sector};
  \end{scope}
  % ============ Pinch1 sector (tadpole on propagator 2) ============
  \begin{scope}[shift={(5.9,0)}]
    \node[bigdot] (t1) at (0,0) {};
    \draw[->,thick,proppurple] (t1) arc[start angle=35,end angle=395,radius=1.05];
    \node[lab] at (0,-1.75) {Pinch$_1$ sector};
  \end{scope}
  % ============ Pinch2 sector (tadpole on propagator 1) ============
  \begin{scope}[shift={(9.9,0)}]
    \node[bigdot] (t2) at (0,0) {};
    \draw[->,thick,propgreen] (t2) arc[start angle=35,end angle=395,radius=1.05];
    \node[lab] at (0,-1.75) {Pinch$_2$ sector};
  \end{scope}
\end{tikzpicture}
\end{document}
