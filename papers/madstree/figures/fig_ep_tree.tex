% Standalone source of the massless three-vertex chain of the third
% application example (Sec. 5, regulator mode), redrawn element by element
% after the original three_vertex_tree figure exactly as fig_tree_diagram, with
% the massless counterparts of its labels: the propagators carry only the
% momentum magnitudes s_1,s_2 (no mass labels, the massless propagators having
% the fixed order \nu=1/2) and the vertices carry the regularized time powers
% a_1,a_2,a_3.
% Compile: pdflatex fig_ep_tree.tex   (included by paper.tex)
\documentclass[tikz,border=6pt]{standalone}
\usetikzlibrary{patterns.meta}
\definecolor{propteal}{RGB}{0,166,172}
\tikzset{
  vtx/.style={circle, draw=black, line width=1.0pt, fill=white,
    postaction={pattern={Lines[angle=45, distance=1.3pt, line width=0.45pt]},
      pattern color=black},
    minimum size=#1, inner sep=0pt, outer sep=0pt},
  vtx/.default=4.2pt,
  leg/.style={draw=black, line width=0.9pt},
  sq/.style={draw=black, line width=0.9pt, fill=white, minimum size=3.0pt,
    inner sep=0pt, outer sep=0pt},
  prop/.style={line width=1.2pt},
}
\begin{document}
\begin{tikzpicture}[every node/.style={inner sep=1.5pt}]
  \coordinate (v1) at (0,0);
  \coordinate (v2) at (2.22,0);
  \coordinate (v3) at (4.54,0);
  % External legs: two per vertex, open squares at the tips.
  \draw[leg] (v1) -- ++(149:1.52) node[sq] {};
  \draw[leg] (v1) -- ++(200:1.45) node[sq] {};
  \draw[leg] (v2) -- ++(120:1.85) node[sq] {};
  \draw[leg] (v2) -- ++(60:1.85)  node[sq] {};
  \draw[leg] (v3) -- ++(31:1.52)  node[sq] {};
  \draw[leg] (v3) -- ++(-20:1.45) node[sq] {};
  % Summed external energies beside the fans.
  \node at (-1.02,1.06) {\small $E_1$};
  \node at (2.22,2.10)  {\small $E_2$};
  \node at (5.56,1.06)  {\small $E_3$};
  % Internal massless propagators: teal lines with momentum labels above.
  \draw[prop,propteal] (v1) -- (v2);
  \draw[prop,propteal] (v2) -- (v3);
  \node[propteal] at (1.11,0.28) {\small $s_1$};
  \node[propteal] at (3.38,0.28) {\small $s_2$};
  % Vertex labels below the vertices.
  \node[anchor=north] at (0,-0.20)    {\small $V_1(\tau_1,a_1)$};
  \node[anchor=north] at (2.22,-0.20) {\small $V_2(\tau_2,a_2)$};
  \node[anchor=north] at (4.54,-0.20) {\small $V_3(\tau_3,a_3)$};
  % Vertices last, on top of the line ends.
  \node[vtx] at (v1) {};
  \node[vtx] at (v2) {};
  \node[vtx] at (v3) {};
\end{tikzpicture}
\end{document}
