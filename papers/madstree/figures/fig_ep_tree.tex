% Standalone source of the massless three-vertex chain of the regulator example
% (Sec. 5). The drawing reuses the massive chain (fig_tree_diagram) with the
% massless counterparts of its parameters: the same external-leg fans with summed
% energies E_v and the same teal propagators carrying momentum labels only, since
% the massless propagators have the fixed order nu = 1/2.
% Compile: pdflatex fig_ep_tree.tex   (included by paper.tex)
\documentclass[tikz,border=6pt]{standalone}
\usetikzlibrary{calc}
\definecolor{propblue}{RGB}{0,161,168}
\begin{document}
\begin{tikzpicture}[every node/.style={inner sep=1.5pt}]
  \coordinate (v1) at (0,0);
  \coordinate (v2) at (4.0,0);
  \coordinate (v3) at (8.0,0);
  % External-leg fans above each vertex; per-leg momentum labels at the tips,
  % summed energies E_v outside the fans.
  \draw (v1) -- (-1.75,1.30);
  \draw (v1) -- (-0.95,1.90);
  \draw (v1) -- (-0.15,2.30);
  \node[anchor=south east] at (-1.75,1.30) {\small $k_{1,1}$};
  \node[anchor=south east] at (-0.95,1.90) {\small $k_{1,2}$};
  \node[anchor=south]      at (-0.15,2.30) {\small $k_{1,3}$};
  \node at (-2.75,1.05) {\small $E_1$};
  \draw (v2) -- (2.75,2.30);
  \draw (v2) -- (3.35,2.55);
  \draw (v2) -- (4.65,2.55);
  \draw (v2) -- (5.25,2.30);
  \node[anchor=south east] at (2.75,2.30) {\small $k_{2,1}$};
  \node[anchor=south]      at (3.35,2.55) {\small $k_{2,2}$};
  \node[anchor=south]      at (4.65,2.55) {\small $k_{2,3}$};
  \node[anchor=south west] at (5.25,2.30) {\small $k_{2,4}$};
  \node at (4.0,3.35) {\small $E_2$};
  \draw (v3) -- (8.15,2.30);
  \draw (v3) -- (8.95,1.90);
  \draw (v3) -- (9.75,1.30);
  \node[anchor=south]      at (8.15,2.30) {\small $k_{3,1}$};
  \node[anchor=south west] at (8.95,1.90) {\small $k_{3,2}$};
  \node[anchor=south west] at (9.75,1.30) {\small $k_{3,3}$};
  \node at (10.75,1.05) {\small $E_3$};
  % Internal massless propagators: teal directed lines with momentum labels
  % above (no mass labels: the order is fixed at nu = 1/2).
  \draw[->,thick,propblue] (v1) -- (v2);
  \draw[->,thick,propblue] (v2) -- (v3);
  \node[propblue] at (2.0,0.34) {\small $s_1$};
  \node[propblue] at (6.0,0.34) {\small $s_2$};
  % Vertex times below the vertices.
  \node[anchor=north] at (0,-0.22)   {\small $\tau_1$};
  \node[anchor=north] at (4.0,-0.22) {\small $\tau_2$};
  \node[anchor=north] at (8.0,-0.22) {\small $\tau_3$};
  % Vertices last.
  \filldraw[fill=black,draw=black] (v1) circle (3pt);
  \filldraw[fill=black,draw=black] (v2) circle (3pt);
  \filldraw[fill=black,draw=black] (v3) circle (3pt);
\end{tikzpicture}
\end{document}
