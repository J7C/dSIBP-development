% Standalone source of the massless three-vertex chain of the regulator example
% (Sec. 5). The drawing reuses the layout of the massive chain (fig_tree_diagram)
% with the massless counterparts of its parameters: the same external energy sums
% E_v and propagator momentum magnitudes s_1,s_2, and no mass labels (the
% massless propagators have the fixed order nu = 1/2).
% Compile: pdflatex fig_ep_tree.tex   (included by paper.tex)
\documentclass[tikz,border=6pt]{standalone}
\usetikzlibrary{calc}
\begin{document}
\begin{tikzpicture}[every node/.style={inner sep=1.5pt}]
  \coordinate (v1) at (0,0);
  \coordinate (v2) at (2.8,0);
  \coordinate (v3) at (5.6,0);
  % External legs, one per vertex, ending on the boundary squares.
  \draw (v1) -- (-1.1,0.95);
  \draw (v2) -- (2.8,1.45);
  \draw (v3) -- (6.7,0.95);
  \filldraw[fill=white,draw=black] ($(-1.1,0.95)+(-0.075,0.075)$) rectangle ++(0.15,-0.15);
  \filldraw[fill=white,draw=black] ($(2.8,1.45)+(-0.075,0.075)$) rectangle ++(0.15,-0.15);
  \filldraw[fill=white,draw=black] ($(6.7,0.95)+(-0.075,0.075)$) rectangle ++(0.15,-0.15);
  \node[left=2pt]  at (-1.1,0.95) {$E_1$};
  \node[right=2pt] at (2.8,1.45)  {$E_2$};
  \node[right=2pt] at (6.7,0.95)  {$E_3$};
  % The two internal massless propagators: directed straight lines, arrows
  % marking the momentum flow, momentum magnitudes above (no mass labels).
  \draw[->,thick] (v1) -- (v2);
  \draw[->,thick] (v2) -- (v3);
  \node[fill=white,inner sep=1pt] at (1.4,0.13) {\small $s_1$};
  \node[fill=white,inner sep=1pt] at (4.2,0.13) {\small $s_2$};
  % Vertex times below the vertices.
  \node[anchor=north] at (0,-0.22)   {\small $\tau_1$};
  \node[anchor=north] at (2.8,-0.22) {\small $\tau_2$};
  \node[anchor=north] at (5.6,-0.22) {\small $\tau_3$};
  % Vertices last.
  \filldraw[fill=black,draw=black] (v1) circle (3pt);
  \filldraw[fill=black,draw=black] (v2) circle (3pt);
  \filldraw[fill=black,draw=black] (v3) circle (3pt);
\end{tikzpicture}
\end{document}
